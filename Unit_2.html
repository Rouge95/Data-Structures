Unit 2: Array-Based Lists

Pseudocode: Implementing a List Interface
  • Define the class: MyList implements List
  • Internal storage:
    • Declare storage (array or linked structure)
    • Declare size = 0
  • add(element):
    • If storage is full, expand storage
    • Insert element at next index
    • Increment size
  • get(index):
    • If index < 0 or >= size, throw error
    • Return element at index
  • remove(index):
    • If index < 0 or >= size, throw error
    • Shift elements after index left by 1
    • Decrement size
  • size():
    • Return current size
  • contains(element):
    • Loop through storage
    • If element matches, return true
    • Else return false
  • Optional methods:
    • isEmpty(): Return size == 0
    • clear(): Remove all elements, set size = 0
    • indexOf(element): Return index if found, else -1
C Implementation (No Stdlib) 

// C: Simple List Implementation (Array-based, no stdlib)
#define MAX_SIZE 100

typedef struct {
    int data[MAX_SIZE];
    int size;
} MyList;

void initList(MyList* list) { list->size = 0; }
void add(MyList* list, int element) {
    if(list->size < MAX_SIZE) { list->data[list->size++] = element; }
}
int get(MyList* list, int index) {
    if(index >= 0 && index < list->size) return list->data[index];
    return -1;
}
void removeAt(MyList* list, int index) {
    if(index >= 0 && index < list->size) {
        for(int i = index; i < list->size - 1; i++) list->data[i] = list->data[i + 1];
        list->size--;
    }
}
int contains(MyList* list, int element) {
    for(int i = 0; i < list->size; i++) if(list->data[i] == element) return 1;
    return 0;
}
int size(MyList* list) { return list->size; }
Java Implementation 

// Java: Simple List Implementation (Array-based)
class MyList {
    private Object[] data;
    private int size;
    private static final int CAPACITY = 100;

    public MyList() { data = new Object[CAPACITY]; size = 0; }

    public void add(E element) { if(size < CAPACITY) data[size++] = element; }

    public E get(int index) {
        if(index >= 0 && index < size) return (E) data[index];
        throw new IndexOutOfBoundsException();
    }

    public void remove(int index) {
        if(index >= 0 && index < size) {
            for(int i = index; i < size - 1; i++) data[i] = data[i + 1];
            size--;
        } else throw new IndexOutOfBoundsException();
    }

    public int size() { return size; }
    public boolean contains(E element) {
        for(int i = 0; i < size; i++) if(data[i].equals(element)) return true;
        return false;
    }
    public void clear() { size = 0; }
    public boolean isEmpty() { return size == 0; }
    public int indexOf(E element) {
        for(int i = 0; i < size; i++) if(data[i].equals(element)) return i;
        return -1;
    }
}
Pseudocode: Implementing a Queue Interface

note: 0 idx front / n idx ; just make frontidx ++ when dequeue

  • Define the class: MyQueue implements Queue
  • Internal storage:
    • Declare storage (array or linked structure)
    • Declare front = 0, rear = 0, size = 0
  • enqueue(element):
    • If storage is full, expand storage
    • Insert element at rear index
    • Increment rear and size
  • dequeue():
    • If size == 0, throw error
    • Return element at front index
    • Increment front and decrement size
  • peek():
    • If size == 0, throw error
    • Return element at front index
  • isEmpty():
    • Return size == 0

Implementation is not circular.

C Implementation (No Stdlib) - Queue 

// C: Simple Queue Implementation (Array-based, no stdlib)
#define MAX_SIZE 100

typedef struct {
    int data[MAX_SIZE];
    int front;
    int rear;
    int size;
} MyQueue;

void initQueue(MyQueue* q) { q->front = q->rear = q->size = 0; }

void enqueue(MyQueue* q, int element) {
    if(q->size < MAX_SIZE) {
        q->data[q->rear] = element;
        q->rear = (q->rear + 1) % MAX_SIZE;
        q->size++;
    }
}

int dequeue(MyQueue* q) {
    if(q->size == 0) return -1; // empty queue
    int element = q->data[q->front];
    q->front = (q->front + 1) % MAX_SIZE;
    q->size--;
    return element;
}

int peek(MyQueue* q) {
    if(q->size == 0) return -1;
    return q->data[q->front];
}

int isEmpty(MyQueue* q) { return q->size == 0; }
Java Implementation - Queue 

// Java: Simple Queue Implementation (Array-based)
class MyQueue {
    private Object[] data;
    private int front;
    private int rear;
    private int size;
    private static final int CAPACITY = 100;

    public MyQueue() { data = new Object[CAPACITY]; front = rear = size = 0; }

    public void enqueue(E element) {
        if(size < CAPACITY) {
            data[rear] = element;
            rear = (rear + 1) % CAPACITY;
            size++;
        }
    }

    public E dequeue() {
        if(size == 0) throw new IllegalStateException("Queue is empty");
        E element = (E) data[front];
        front = (front + 1) % CAPACITY;
        size--;
        return element;
    }

    public E peek() {
        if(size == 0) throw new IllegalStateException("Queue is empty");
        return (E) data[front];
    }

    public boolean isEmpty() { return size == 0; }
}
Pseudocode: Implementing a Stack Interface
  • Define the class: MyStack implements Stack
  • Internal storage:
    • Declare storage (array or linked structure)
    • Declare top = -1 (empty stack)
  • push(element):
    • If storage is full, expand storage
    • Increment top
    • Insert element at top index
  • pop():
    • If top == -1, throw error (stack empty)
    • Return element at top index
    • Decrement top
  • peek():
    • If top == -1, throw error
    • Return element at top index
  • isEmpty():
    • Return true if top == -1
  • size():
    • Return top + 1
C Implementation (No Stdlib) - Stack 

        // C: Simple Stack Implementation (Array-based, no stdlib)
        #define MAX_SIZE 100
        
        typedef struct {
            int data[MAX_SIZE];
            int top;
        } MyStack;
        
        void initStack(MyStack* s) { s->top = -1; }
        
        void push(MyStack* s, int element) {
            if(s->top < MAX_SIZE - 1) {
                s->top++;
                s->data[s->top] = element;
            }
        }
        
        int pop(MyStack* s) {
            if(s->top == -1) return -1; // stack empty
            int element = s->data[s->top];
            s->top--;
            return element;
        }
        
        int peek(MyStack* s) {
            if(s->top == -1) return -1;
            return s->data[s->top];
        }
        
        int isEmpty(MyStack* s) { return s->top == -1; }
        
        int size(MyStack* s) { return s->top + 1; }
Java Implementation - Stack 

        // Java: Simple Stack Implementation (Array-based)
        class MyStack {
            private Object[] data;
            private int top;
            private static final int CAPACITY = 100;
        
            public MyStack() {
                data = new Object[CAPACITY];
                top = -1;
            }
        
            public void push(E element) {
                if(top < CAPACITY - 1) {
                    top++;
                    data[top] = element;
                }
            }
        
            public E pop() {
                if(top == -1) throw new IllegalStateException("Stack is empty");
                E element = (E) data[top];
                top--;
                return element;
            }
        
            public E peek() {
                if(top == -1) throw new IllegalStateException("Stack is empty");
                return (E) data[top];
            }
        
            public boolean isEmpty() { return top == -1; }
        
            public int size() { return top + 1; }
        }
